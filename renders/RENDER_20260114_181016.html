javascript:(function(){
  let el =
    document.querySelector('article [data-message-author-role="assistant"]') ||
    document.querySelector('[data-testid="conversation-turn"]:last-child') ||
    document.querySelector('article');

  if(!el){
    alert("❌ No ChatGPT content found");
    return;
  }

  fetch("http://localhost:8765",{
    method:"POST",
    headers:{ "Content-Type":"application/json" },
    body: JSON.stringify({
      html: el.innerHTML,
      source: "ChatGPT"
    })
  })
  .then(r=>r.json())
  .then(d=>alert("✅ Captured"))
  .catch(()=>alert("❌ Server not running"));
})();
