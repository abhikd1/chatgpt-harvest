javascript:(function(){
try{
let html='';
let source='Unknown';

/* 1️⃣ Try user selection first */
let sel=window.getSelection();
if(sel && sel.rangeCount>0 && sel.toString().trim().length>0){
 let r=sel.getRangeAt(0);
 let d=document.createElement('div');
 d.appendChild(r.cloneContents());
 html=d.innerHTML;
}

/* 2️⃣ ChatGPT (current UI) */
if(!html){
 let blocks=document.querySelectorAll('article div[data-message-author-role="assistant"]');
 if(blocks.length){
  html=[...blocks].map(b=>b.innerHTML).join('<hr>');
  source='ChatGPT';
 }
}

/* 3️⃣ Claude */
if(!html){
 let c=document.querySelectorAll('[data-test-render-count]');
 if(c.length){
  html=[...c].map(e=>e.innerHTML).join('<hr>');
  source='Claude';
 }
}

/* 4️⃣ Gemini */
if(!html){
 let g=document.querySelectorAll('.model-response-text, .response-content');
 if(g.length){
  html=[...g].map(e=>e.innerHTML).join('<hr>');
  source='Gemini';
 }
}

/* 5️⃣ Final fallback: visible main content */
if(!html){
 let main=document.querySelector('main');
 if(main){
  html=main.innerHTML;
  source='PageFallback';
 }
}

if(!html || html.trim().length<20){
 alert('❌ No content detected.\n\nScroll so the AI reply is visible and try again.');
 return;
}

/* SEND TO SERVER */
fetch('http://localhost:8765',{
 method:'POST',
 headers:{'Content-Type':'application/json'},
 body:JSON.stringify({html:html,source:source})
})
.then(r=>r.json())
.then(d=>{
 if(d.status==='success'){
  alert('✅ Captured!\n\n'+d.message);
 }else{
  alert('❌ Failed:\n\n'+d.message);
 }
})
.catch(()=>{
 alert('❌ Server not running!\n\nRun: python capture_server.py');
});

}catch(e){
 alert('❌ Capture error:\n'+e.message);
}
})();

  1. Click Save
