if(!html){
 let blocks=document.querySelectorAll('article div[data-message-author-role="assistant"]');

/* 3️⃣ Claude */
if(!html){
 let c=document.querySelectorAll('[data-test-render-count]');

  source='Claude';
 }
}

/* 4️⃣ Gemini */
if(!html){
 let g=document.querySelectorAll('.model-response-text, .response-content');

/* 5️⃣ Final fallback: visible main content */
if(!html){
 let main=document.querySelector('main');
 if(main){

if(!html || html.trim().length<20){

 return;

 headers:{'Content-Type':'application/json'},
